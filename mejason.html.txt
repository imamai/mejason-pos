function addToCart(id) {
  const svc = SERVICES.find(s => s.id === id);
  if(!svc) return;
  const existing = cart.find(c => c.id === id);
  if(existing) { existing.qty++; } else { cart.push({...svc, qty:1}); }
  renderCart();
}

function removeFromCart(id) {
  cart = cart.filter(c => c.id !== id);
  renderCart();
}

function recalc() {
  const sub = cart.reduce((a,c) => a + c.price * c.qty, 0);
  const disc = Math.min(parseFloat(document.getElementById('discount-pct').value)||0, 100);
  const discAmt = sub * disc / 100;
  const total = sub - discAmt;
  document.getElementById('subtotal').textContent = fmt(sub);
  document.getElementById('discount-amt').textContent = '- ' + fmt(discAmt);
  document.getElementById('cart-total').textContent = fmt(total);
  return {sub, discAmt, total};
}

function renderCart() {
  const el = document.getElementById('cart-items');
  document.getElementById('cart-count').textContent = cart.length + ' item' + (cart.length !== 1 ? 's' : '');
  if(!cart.length) { el.innerHTML = '<div style="color:var(--muted);font-size:12px;text-align:center;padding:20px">Add services to get started</div>'; recalc(); return; }
  el.innerHTML = cart.map(c => `
    <div class="cart-item">
      <div class="cart-item-name">${c.name}<br><span style="font-size:10px;color:var(--muted)">Qty: ${c.qty}</span></div>
      <div class="cart-item-price">${fmt(c.price * c.qty)}</div>
      <span class="cart-remove" onclick="removeFromCart('${c.id}')">×</span>
    </div>`).join('');
  recalc();
}

function checkout() {
  const client = document.getElementById('cart-client').value;
  if(!client) { alert('Please select a client.'); return; }
  if(!cart.length) { alert('Cart is empty.'); return; }
  const {sub, discAmt, total} = recalc();
  const disc = parseFloat(document.getElementById('discount-pct').value)||0;
  const eventDate = document.getElementById('event-date').value;
  const invNum = 'MMP-' + String(invoices.length + 300 + 1).padStart(3,'0');
  const inv = {
    id: invNum, client, items: [...cart], subtotal: sub,
    discountPct: disc, discountAmt: discAmt, total,
    payMethod, status: payMethod === 'credit' ? 'Unpaid' : 'Paid',
    date: new Date().toLocaleDateString('en-KE'), eventDate
  };
  invoices.push(inv);
  showInvoice(inv);
  cart = [];
  document.getElementById('discount-pct').value = '';
  renderCart();
}

function renderDashboard() {
  const total = invoices.reduce((a,i) => a + i.total, 0);
  const unpaid = invoices.filter(i => i.status === 'Unpaid').reduce((a,i) => a + i.total, 0);
  document.getElementById('s-rev').textContent = fmt(total);
  document.getElementById('s-unpaid').textContent = fmt(unpaid);
  document.getElementById('s-tx').textContent = invoices.length;

  const tbody = document.getElementById('recent-tx');
  if(!invoices.length) { tbody.innerHTML = '<tr><td colspan="4" style="color:var(--muted);text-align:center;padding:20px">No transactions yet</td></tr>'; }
  else {
    tbody.innerHTML = [...invoices].reverse().slice(0,5).map(i => `
      <tr>
        <td>${i.client}</td>
        <td style="color:var(--muted);font-size:11px">${i.items.map(x=>x.name).join(', ').substring(0,40)}…</td>
        <td style="font-weight:600">${fmt(i.total)}</td>
        <td><span class="badge ${i.status==='Paid'?'badge-green':'badge-red'}">${i.status}</span></td>
      </tr>`).join('');
  }

  const cats = {};
  invoices.forEach(inv => inv.items.forEach(item => { cats[item.cat] = (cats[item.cat]||0) + item.price*item.qty; }));
  const catNames = {wedding:'Wedding',traditional:'Traditional',studio:'Studio',editing:'Editing',equipment:'Equipment'};
  const colors = {wedding:'#E31E24',traditional:'#f59e0b',studio:'#3b82f6',editing:'#22c55e',equipment:'#8b5cf6'};
  const catEl = document.getElementById('cat-breakdown');
  const entries = Object.entries(cats);
  if(!entries.length) { catEl.innerHTML = '<div style="color:var(--muted);font-size:12px;text-align:center;padding:20px">No data yet</div>'; return; }
  const max = Math.max(...entries.map(e=>e[1]));
  catEl.innerHTML = entries.map(([cat,val]) => `
    <div style="margin-bottom:10px">
      <div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:4px"><span>${catNames[cat]||cat}</span><span style="color:var(--muted)">${fmt(val)}</span></div>
      <div style="background:var(--border);border-radius:4px;height:6px"><div style="background:${colors[cat]||'#E31E24'};width:${Math.round(val/max*100)}%;height:6px;border-radius:4px"></div></div>
    </div>`).join('');
}

function renderReports() {
  const total = invoices.reduce((a,i) => a + i.total, 0);
  const unpaid = invoices.filter(i => i.status === 'Unpaid').reduce((a,i) => a + i.total, 0);
  document.getElementById('r-sales').textContent = fmt(total);
  document.getElementById('r-unpaid').textContent = fmt(unpaid);
  document.getElementById('r-paid').textContent = fmt(total - unpaid);
  document.getElementById('r-tx').textContent = invoices.length;

  const cats = {};
  invoices.forEach(inv => inv.items.forEach(item => { cats[item.cat] = (cats[item.cat]||0) + item.price*item.qty; }));
  const catNames = {wedding:'Wedding',traditional:'Traditional',studio:'Studio',editing:'Editing',equipment:'Equipment'};
  const all = Object.entries(cats);
  const max = all.length ? Math.max(...all.map(e=>e[1])) : 1;
  const barsEl = document.getElementById('report-bars');
  barsEl.innerHTML = all.length ? all.map(([cat,val]) => `
    <div style="margin-bottom:12px">
      <div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:4px"><span>${catNames[cat]||cat}</span><span style="color:var(--muted)">${fmt(val)}</span></div>

    </div>`).join('') : '<div style="color:var(--muted);font-size:12px;text-align:center;padding:20px">No data yet — complete a sale</div>';

  const clientRev = {};
  invoices.forEach(inv => { clientRev[inv.client] = (clientRev[inv.client]||0) + inv.total; });
  const topEl = document.getElementById('top-clients');
  const sorted = Object.entries(clientRev).sort((a,b) => b[1]-a[1]).slice(0,5);
  topEl.innerHTML = sorted.length ? sorted.map(([name,val],idx) => `
    <div style="display:flex;align-items:center;gap:10px;padding:8px 0;border-bottom:1px solid rgba(255,255,255,.05)">
      <div style="width:24px;height:24px;border-radius:50%;background:var(--red);display:flex;align-items:center;justify-content:center;font-size:11px;font-weight:700;color:#fff;flex-shrink:0">${idx+1}</div>

      <div style="font-size:13px;font-weight:600">${fmt(val)}</div>
    </div>`).join('') : '<div style="color:var(--muted);font-size:12px;text-align:center;padding:20px">No data yet</div>';
}
